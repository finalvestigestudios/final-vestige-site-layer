<script id="fv-listen-mode-r1-js">
(()=>{if(window.__fvListenMode)return;window.__fvListenMode=true;
const body=document.body;if(!body||(!body.classList.contains('single-post')&&!body.classList.contains('single')))return;
if(!('speechSynthesis'in window)||!('SpeechSynthesisUtterance'in window))return;
const main=document.querySelector('main')||document;const article=main.querySelector('article')||main;
const sourceRoot=article.querySelector('[data-fv-footnotes],.fv-source-list')||document.querySelector('[data-fv-footnotes],.fv-source-list');
if(!article||!sourceRoot)return;
const beforeSource=node=>Boolean(node.compareDocumentPosition(sourceRoot)&Node.DOCUMENT_POSITION_FOLLOWING);
const featureUi=node=>Boolean(node.closest('.fv-listen-mode,.fv-case-search,.fv-passage-kit,.fv-passage-kit-action,.fv-return-trail,.fv-reading-focus,.fv-evidence-trace,.fv-source-kit,.fv-chronology-rail,.fv-evidence-map,.fv-detail-lens'));
const clean=s=>String(s||'').replace(/\s+/g,' ').trim();
const spokenText=node=>{const clone=node.cloneNode(true);clone.querySelectorAll('.fv-fn-ref[data-fv-fn],.fv-fn-mark,.fv-fn-backlinks,.fv-passage-kit-action,[aria-hidden="true"]').forEach(n=>n.remove());return clean(clone.textContent)};
const prose=[...article.querySelectorAll('p')].filter(p=>beforeSource(p)&&!featureUi(p)&&!p.closest('figure,figcaption,header,footer,nav,form,.sharedaddy,.sd-sharing-enabled,.wp-block-buttons')&&spokenText(p).length>=60);
const totalText=prose.reduce((n,p)=>n+spokenText(p).length,0);
if(prose.length<12||totalText<3000)return;
const headings=[...article.querySelectorAll('h2,h3')].filter(h=>beforeSource(h)&&!featureUi(h));
const sectionFor=block=>{let label='Opening';for(const h of headings){if(h===block)break;if(h.compareDocumentPosition(block)&Node.DOCUMENT_POSITION_FOLLOWING)label=clean(h.textContent)}return label};
const segments=[];
const storyNodes=[...headings,...prose].sort((a,b)=>a===b?0:(a.compareDocumentPosition(b)&Node.DOCUMENT_POSITION_FOLLOWING?-1:1));
storyNodes.forEach(node=>{const text=spokenText(node);if(text)segments.push({el:node,text,kind:node.matches('h2,h3')?'Section':'Passage',section:node.matches('h2,h3')?text:sectionFor(node)})});
if(segments.length<12)return;
const launcher=document.createElement('section');launcher.className='fv-listen-mode';launcher.setAttribute('role','region');launcher.setAttribute('aria-label','Listen Mode');
launcher.innerHTML=`<div class="fv-listen-mode__intro"><div><p class="fv-listen-mode__kicker">LISTEN MODE</p><p class="fv-listen-mode__title">Listen to the investigation without losing your place.</p><p class="fv-listen-mode__note">Uses the speech engine provided by your browser or device. Final Vestige does not send article text to a speech service.</p></div><button class="fv-listen-mode__open" type="button">Listen to this investigation</button></div><div class="fv-listen-mode__player" aria-label="Investigation narration controls"><div class="fv-listen-mode__now"><p class="fv-listen-mode__status">Ready to listen<span class="fv-listen-mode__section">Narrative only · source notes stay on the page</span></p><span class="fv-listen-mode__position">1 of ${segments.length}</span></div><div class="fv-listen-mode__controls"><button class="fv-listen-mode__control" type="button" data-action="prev" aria-label="Previous passage">Previous</button><button class="fv-listen-mode__control" type="button" data-action="play">Play</button><button class="fv-listen-mode__control" type="button" data-action="next" aria-label="Next passage">Next</button><button class="fv-listen-mode__control" type="button" data-action="stop">Close</button><div class="fv-listen-mode__rates" aria-label="Narration speed"><button class="fv-listen-mode__rate" type="button" data-rate="0.8" aria-pressed="false">0.8×</button><button class="fv-listen-mode__rate" type="button" data-rate="1" aria-pressed="true">1×</button><button class="fv-listen-mode__rate" type="button" data-rate="1.25" aria-pressed="false">1.25×</button><button class="fv-listen-mode__rate" type="button" data-rate="1.5" aria-pressed="false">1.5×</button></div></div><p class="fv-listen-mode__sr" role="status" aria-live="polite"></p></div>`;
const firstHeading=headings[0];if(firstHeading)firstHeading.before(launcher);else prose[Math.min(2,prose.length-1)].after(launcher);
const open=launcher.querySelector('.fv-listen-mode__open');const player=launcher.querySelector('.fv-listen-mode__player');const play=launcher.querySelector('[data-action="play"]');const prev=launcher.querySelector('[data-action="prev"]');const next=launcher.querySelector('[data-action="next"]');const stop=launcher.querySelector('[data-action="stop"]');const status=launcher.querySelector('.fv-listen-mode__status');const position=launcher.querySelector('.fv-listen-mode__position');const live=launcher.querySelector('[role="status"]');const rateButtons=[...launcher.querySelectorAll('.fv-listen-mode__rate')];
const synth=window.speechSynthesis;let current=0;let rate=1;let utterance=null;let token=0;let paused=false;let active=false;let completed=false;let highlighted=null;let highlightOwned=false;let highlightHadClassAttr=false;
const announce=text=>{live.textContent='';requestAnimationFrame(()=>live.textContent=text)};
const clearHighlight=()=>{if(!highlighted)return;if(highlightOwned){highlighted.classList.remove('fv-listen-mode-current');if(!highlightHadClassAttr&&!highlighted.getAttribute('class'))highlighted.removeAttribute('class')}highlighted=null;highlightOwned=false;highlightHadClassAttr=false};
const markHighlight=el=>{clearHighlight();highlighted=el;highlightHadClassAttr=el.hasAttribute('class');highlightOwned=!el.classList.contains('fv-listen-mode-current');if(highlightOwned)el.classList.add('fv-listen-mode-current')};
const showCurrent=(scroll=false)=>{const item=segments[current];markHighlight(item.el);status.firstChild.nodeValue=completed?'Finished':paused?'Paused':synth.speaking?'Playing':'Ready';status.querySelector('.fv-listen-mode__section').textContent=`${item.kind} · ${item.section}`;position.textContent=`${current+1} of ${segments.length}`;play.textContent=paused?'Resume':synth.speaking?'Pause':'Play';prev.disabled=current===0;next.disabled=current===segments.length-1;if(scroll){const r=item.el.getBoundingClientRect();if(r.top<120||r.bottom>innerHeight-80)item.el.scrollIntoView({behavior:matchMedia('(prefers-reduced-motion: reduce)').matches?'auto':'smooth',block:'center'})}};
const cancel=()=>{token++;utterance=null;paused=false;synth.cancel()};
const speakCurrent=()=>{cancel();completed=false;active=true;launcher.classList.add('is-active');const id=token;const item=segments[current];utterance=new SpeechSynthesisUtterance(item.text);utterance.lang=document.documentElement.lang||'en-US';utterance.rate=rate;utterance.onstart=()=>{if(id!==token)return;paused=false;showCurrent(true);announce(`Playing ${item.kind.toLowerCase()} ${current+1} of ${segments.length}.`)};utterance.onend=()=>{if(id!==token)return;if(current<segments.length-1){current++;speakCurrent()}else{utterance=null;completed=true;paused=false;showCurrent(false);announce('Narration finished.')}};utterance.onerror=e=>{if(id!==token||e.error==='canceled'||e.error==='interrupted')return;utterance=null;paused=false;showCurrent(false);announce('Narration stopped because the browser speech engine reported an error.')};synth.speak(utterance);showCurrent(false)};
const openPlayer=()=>{active=true;completed=false;launcher.classList.add('is-active');showCurrent(false);play.focus();announce('Listen Mode ready. Press Play to begin.')};
const closePlayer=()=>{cancel();active=false;completed=false;launcher.classList.remove('is-active');clearHighlight();open.focus();announce('Listen Mode closed.')};
open.addEventListener('click',openPlayer);
play.addEventListener('click',()=>{if(paused){synth.resume();paused=false;showCurrent(false);announce('Narration resumed.');return}if(synth.speaking){synth.pause();paused=true;showCurrent(false);announce('Narration paused.');return}speakCurrent()});
prev.addEventListener('click',()=>{if(current===0)return;const wasPlaying=synth.speaking&&!paused;current--;cancel();showCurrent(true);if(wasPlaying)speakCurrent();else announce(`Moved to ${segments[current].kind.toLowerCase()} ${current+1}.`)});
next.addEventListener('click',()=>{if(current>=segments.length-1)return;const wasPlaying=synth.speaking&&!paused;current++;cancel();showCurrent(true);if(wasPlaying)speakCurrent();else announce(`Moved to ${segments[current].kind.toLowerCase()} ${current+1}.`)});
stop.addEventListener('click',closePlayer);
rateButtons.forEach(button=>button.addEventListener('click',()=>{rate=Number(button.dataset.rate)||1;rateButtons.forEach(b=>b.setAttribute('aria-pressed',String(b===button)));const wasPlaying=synth.speaking||paused;if(wasPlaying)speakCurrent();announce(`Narration speed ${button.textContent}.${wasPlaying?' Current passage restarted.':''}`)}));
document.addEventListener('keydown',e=>{if(active&&e.key==='Escape'){e.preventDefault();closePlayer()}});
window.addEventListener('pagehide',()=>{token++;synth.cancel();clearHighlight()});
})();
</script>
