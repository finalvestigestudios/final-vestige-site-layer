<script id="fv-detail-lens-script">
(()=>{
  if(window.__fvDetailLens)return;
  window.__fvDetailLens=true;
  const supported=typeof HTMLDialogElement!=='undefined'&&typeof HTMLDialogElement.prototype.showModal==='function';
  if(!supported)return;
  const ready=fn=>document.readyState==='loading'?document.addEventListener('DOMContentLoaded',fn,{once:true}):fn();
  ready(()=>{
    if(!document.body.classList.contains('single-post'))return;
    const dialog=document.getElementById('fv-detail-lens');
    if(!dialog)return;
    const root=[...document.querySelectorAll('main .wp-block-post-content,main article .entry-content,main article')].find(el=>el.querySelector('figure.wp-block-image img'))||document.querySelector('main');
    if(!root)return;
    const clean=s=>(s||'').replace(/\s+/g,' ').trim();
    const captionText=fig=>{const cap=fig.querySelector('figcaption');if(!cap)return'';const clone=cap.cloneNode(true);clone.querySelectorAll('.fv-image-provenance').forEach(el=>el.remove());return clean(clone.textContent)};
    const bestSrc=img=>{
      const set=img.getAttribute('srcset')||'';
      const candidates=set.split(',').map(part=>{const bits=part.trim().split(/\s+/);const w=bits[1]&&/^[0-9]+w$/.test(bits[1])?parseInt(bits[1],10):0;return{src:bits[0],w}}).filter(x=>x.src);
      if(candidates.some(x=>x.w))return candidates.sort((a,b)=>b.w-a.w)[0].src;
      return img.currentSrc||img.getAttribute('src')||'';
    };
    const records=[...root.querySelectorAll('figure.wp-block-image')].map((figure,index)=>{
      if(figure.closest('header,footer,nav'))return null;
      const img=figure.querySelector('img');
      if(!img||img.classList.contains('avatar')||img.width<80&&img.height<80)return null;
      const source=figure.querySelector('.fv-image-provenance a[href]');
      return{figure,img,index,src:bestSrc(img),alt:clean(img.getAttribute('alt')),caption:captionText(figure),source:source?source.href:''};
    }).filter(Boolean).map((r,i)=>({...r,index:i}));
    if(!records.length)return;

    const stage=dialog.querySelector('[data-fv-detail-stage]');
    const view=dialog.querySelector('[data-fv-detail-image]');
    const status=dialog.querySelector('[data-fv-detail-status]');
    const count=dialog.querySelector('[data-fv-detail-count]');
    const caption=dialog.querySelector('[data-fv-detail-caption]');
    const desc=dialog.querySelector('[data-fv-detail-desc]');
    const original=dialog.querySelector('[data-fv-detail-original]');
    const source=dialog.querySelector('[data-fv-detail-source]');
    const close=dialog.querySelector('[data-fv-detail-close]');
    const prev=dialog.querySelector('[data-fv-detail-prev]');
    const next=dialog.querySelector('[data-fv-detail-next]');
    const zin=dialog.querySelector('[data-fv-detail-in]');
    const zout=dialog.querySelector('[data-fv-detail-out]');
    const fit=dialog.querySelector('[data-fv-detail-fit]');
    const zoomLabel=dialog.querySelector('[data-fv-detail-zoom]');
    const zooms=[1,1.25,1.6,2,2.6,3.4,4.5,6];
    let active=0,zoomIndex=0,baseW=0,baseH=0,invoker=null,loadToken=0;

    const updateZoomUI=()=>{
      zoomLabel.textContent=zoomIndex===0?'Fit':`${Math.round(zooms[zoomIndex]*100)}%`;
      zout.disabled=zoomIndex===0;
      zin.disabled=zoomIndex===zooms.length-1;
    };
    const computeBase=()=>{
      if(!view.naturalWidth||!view.naturalHeight)return;
      const availW=Math.max(120,stage.clientWidth-24),availH=Math.max(120,stage.clientHeight-24);
      const scale=Math.min(availW/view.naturalWidth,availH/view.naturalHeight,1);
      baseW=Math.max(1,view.naturalWidth*scale);baseH=Math.max(1,view.naturalHeight*scale);
    };
    const applySize=(preserveCenter=false)=>{
      if(!baseW||!baseH)return;
      const oldW=Math.max(stage.scrollWidth,1),oldH=Math.max(stage.scrollHeight,1);
      const cx=(stage.scrollLeft+stage.clientWidth/2)/oldW,cy=(stage.scrollTop+stage.clientHeight/2)/oldH;
      view.style.width=`${Math.round(baseW*zooms[zoomIndex])}px`;
      view.style.height=`${Math.round(baseH*zooms[zoomIndex])}px`;
      updateZoomUI();
      requestAnimationFrame(()=>{
        if(preserveCenter){stage.scrollLeft=Math.max(0,cx*stage.scrollWidth-stage.clientWidth/2);stage.scrollTop=Math.max(0,cy*stage.scrollHeight-stage.clientHeight/2)}
        else{stage.scrollLeft=Math.max(0,(stage.scrollWidth-stage.clientWidth)/2);stage.scrollTop=Math.max(0,(stage.scrollHeight-stage.clientHeight)/2)}
      });
    };
    const fitImage=()=>{zoomIndex=0;computeBase();applySize(false)};
    const setZoom=delta=>{const nextIndex=Math.max(0,Math.min(zooms.length-1,zoomIndex+delta));if(nextIndex===zoomIndex)return;zoomIndex=nextIndex;applySize(true)};
    const render=i=>{
      active=Math.max(0,Math.min(records.length-1,i));
      const rec=records[active],token=++loadToken;
      count.textContent=`Figure ${active+1} of ${records.length}`;
      caption.textContent=rec.caption||'No caption is attached to this image.';
      if(rec.alt){desc.hidden=false;desc.innerHTML='<strong>Image description:</strong> '+rec.alt.replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]))}
      else{desc.hidden=true;desc.textContent=''}
      original.href=rec.src;
      if(rec.source){source.href=rec.source;source.hidden=false}else{source.hidden=true;source.removeAttribute('href')}
      prev.disabled=active===0;next.disabled=active===records.length-1;
      status.hidden=false;status.textContent='Loading detail…';
      view.hidden=true;view.alt=rec.alt;view.removeAttribute('src');view.style.width='';view.style.height='';
      zoomIndex=0;baseW=0;baseH=0;updateZoomUI();
      view.onload=()=>{if(token!==loadToken)return;status.hidden=true;view.hidden=false;fitImage()};
      view.onerror=()=>{if(token!==loadToken)return;view.hidden=true;status.hidden=false;status.textContent='This image could not be opened at a larger size.'};
      view.src=rec.src;
    };
    const openAt=(i,button)=>{
      invoker=button;
      document.body.classList.add('fv-detail-lens-open');
      dialog.showModal();
      render(i);
      close.focus({preventScroll:true});
    };
    records.forEach((rec,i)=>{
      rec.figure.classList.add('fv-detail-ready');
      const button=document.createElement('button');button.type='button';button.className='fv-detail-trigger';button.textContent='Inspect detail';button.setAttribute('aria-label',`Inspect figure ${i+1} in Detail Lens`);
      button.addEventListener('click',()=>openAt(i,button));rec.figure.appendChild(button);
    });
    close.addEventListener('click',()=>dialog.close());
    prev.addEventListener('click',()=>{if(active>0)render(active-1)});
    next.addEventListener('click',()=>{if(active<records.length-1)render(active+1)});
    zin.addEventListener('click',()=>setZoom(1));zout.addEventListener('click',()=>setZoom(-1));fit.addEventListener('click',fitImage);
    dialog.addEventListener('close',()=>{document.body.classList.remove('fv-detail-lens-open');if(invoker&&document.contains(invoker))invoker.focus({preventScroll:true})});
    dialog.addEventListener('click',e=>{if(e.target===dialog)dialog.close()});
    dialog.addEventListener('keydown',e=>{
      if(e.key==='+'||e.key==='='){e.preventDefault();setZoom(1)}
      else if(e.key==='-'){e.preventDefault();setZoom(-1)}
      else if(e.key==='0'){e.preventDefault();fitImage()}
      else if(e.key==='['&&active>0){e.preventDefault();render(active-1)}
      else if(e.key===']'&&active<records.length-1){e.preventDefault();render(active+1)}
    });
    let resizeTimer=0;window.addEventListener('resize',()=>{if(!dialog.open||!view.naturalWidth)return;clearTimeout(resizeTimer);resizeTimer=setTimeout(()=>{const z=zoomIndex;computeBase();zoomIndex=z;applySize(true)},80)},{passive:true});
  });
})();
</script>
