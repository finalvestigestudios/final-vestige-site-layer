<script id="fv-visual-trail-v1-js">
(()=>{'use strict';
if(window.__fvVisualTrail)return;window.__fvVisualTrail=true;
const canonical=[...document.querySelectorAll('article h2#sources.fv-source-heading')];
if(canonical.length!==1)return;
const sourceHeading=canonical[0],article=sourceHeading.closest('article');
if(!article)return;
const noteRoots=[...article.querySelectorAll('[data-fv-footnotes]')];
if(noteRoots.length!==1)return;
const notesRoot=noteRoots[0];
if(!(sourceHeading.compareDocumentPosition(notesRoot)&Node.DOCUMENT_POSITION_FOLLOWING))return;
const before=(a,b)=>Boolean(a.compareDocumentPosition(b)&Node.DOCUMENT_POSITION_FOLLOWING);
const storyHeadings=[...article.querySelectorAll('h2')].filter(h=>h!==sourceHeading&&before(h,sourceHeading));
if(storyHeadings.length<2||storyHeadings.length>10)return;
const sectionFor=el=>{
  let label='Opening';
  for(const h of storyHeadings){
    if(before(h,el))label=(h.textContent||'').trim()||'Untitled section';
    else break;
  }
  return label;
};
const figures=[...article.querySelectorAll('figure.wp-block-image')].filter(f=>before(f,sourceHeading)).map((figure,index)=>{
  const img=figure.querySelector('img');if(!img)return null;
  const raw=(img.getAttribute('src')||'').trim();if(!raw)return null;
  let resolved;try{resolved=new URL(raw,location.href)}catch(_){return null}
  if(!/^https?:$/.test(resolved.protocol))return null;
  const alt=(img.getAttribute('alt')||'').trim();
  const caption=((figure.querySelector('figcaption')||{}).textContent||'').trim();
  if(!alt&&!caption)return null;
  return{figure,img,src:resolved.href,alt,caption,section:sectionFor(figure),order:index+1};
}).filter(Boolean);
if(figures.length<3||figures.length>12)return;
const firstStory=storyHeadings[0];
if(!firstStory)return;

const band=document.createElement('section');
band.className='fvvt-band';band.dataset.fvvtBand='';
band.setAttribute('aria-label','Visual Trail');
band.innerHTML='<div class="fvvt-band-copy"><span class="fvvt-kicker">Visual trail</span><strong class="fvvt-band-title"></strong><span class="fvvt-band-note">See the investigation through the images and diagrams already published here.</span></div><button type="button" class="fvvt-button fvvt-open">Follow the visual trail</button>';

const opener=band.querySelector('.fvvt-open');

let dialog=null,index=0,returnToOpener=true;

  const d=document.createElement('dialog');
  d.className='fvvt-dialog';d.setAttribute('aria-labelledby','fvvt-title');d.setAttribute('aria-describedby','fvvt-disclosure');
  d.innerHTML='<div class="fvvt-shell"><header class="fvvt-header"><div class="fvvt-header-copy"><span class="fvvt-kicker">Visual trail</span><h2 class="fvvt-title" id="fvvt-title">Follow the published visuals</h2><p class="fvvt-disclosure" id="fvvt-disclosure">A sequence of the images and diagrams already published in this investigation.   const nav=d.querySelector('.fvvt-index');
  figures.forEach((item,i)=>{
    const b=document.createElement('button');
    b.type='button';b.className='fvvt-index-button';b.textContent=String(i+1).padStart(2,'0');
    b.setAttribute('aria-label',`Go to figure ${i+1}: ${trimLabel(item.section)}`);
    b.addEventListener('click',()=>show(i));
    nav.appendChild(b);
  });
  d.querySelector('.fvvt-close').addEventListener('click',()=>d.close());
  d.querySelector('.fvvt-prev').addEventListener('click',()=>show(Math.max(0,index-1)));
  d.querySelector('.fvvt-next').addEventListener('click',()=>show(Math.min(figures.length-1,index+1)));
  d.querySelector('.fvvt-view').addEventListener('click',viewInStory);
  d.addEventListener('keydown',e=>{
    if(e.key==='ArrowLeft'){e.preventDefault();show(Math.max(0,index-1))}
    else if(e.key==='ArrowRight'){e.preventDefault();show(Math.min(figures.length-1,index+1))}
    else if(e.key==='Home'){e.preventDefault();show(0)}
    else if(e.key==='End'){e.preventDefault();show(figures.length-1)}
  });
  d.addEventListener('close',()=>{if(returnToOpener)setTimeout(()=>opener.focus(),75);returnToOpener=true});
  document.body.appendChild(d);return d;
}
function show(nextIndex){
  if(!dialog)return;
  index=Math.max(0,Math.min(figures.length-1,nextIndex));
  const item=figures[index],image=dialog.querySelector('.fvvt-image');
  if(image.src!==item.src)image.src=item.src;
  image.alt=item.alt||item.caption||`Published figure ${index+1}`;
  dialog.querySelector('.fvvt-position').textContent=`Figure ${index+1} of ${figures.length}`;
  dialog.querySelector('.fvvt-section').textContent=item.section;
  dialog.querySelector('.fvvt-caption').textContent=item.caption||item.alt;
  dialog.querySelector('.fvvt-status').textContent=`${index+1} / ${figures.length}`;
  dialog.querySelector('.fvvt-prev').disabled=index===0;
  dialog.querySelector('.fvvt-next').disabled=index===figures.length-1;
  dialog.querySelectorAll('.fvvt-index-button').forEach((b,i)=>{if(i===index)b.setAttribute('aria-current','true');else b.removeAttribute('aria-current')});
  const stage=dialog.querySelector('.fvvt-stage');stage.setAttribute('aria-label',`Investigation visual trail, figure ${index+1} of ${figures.length}`);
}
function openTrail(){
  if(!dialog)dialog=buildDialog();
  index=0;show(0);returnToOpener=true;dialog.showModal();
  requestAnimationFrame(()=>dialog.querySelector('.fvvt-close').focus());
}
function viewInStory(){
  const item=figures[index],target=item.figure;
  returnToOpener=false;dialog.close();
  const hadTab=target.hasAttribute('tabindex'),oldTab=target.getAttribute('tabindex');
  if(!hadTab)target.setAttribute('tabindex','-1');
  target.classList.add('fvvt-story-focus');
  setTimeout(()=>{
    target.scrollIntoView({behavior:motionOK()?'smooth':'auto',block:'center'});
    target.focus({preventScroll:true});
    const cleanup=()=>{target.classList.remove('fvvt-story-focus');if(!hadTab)target.removeAttribute('tabindex');else if(oldTab!==null)target.setAttribute('tabindex',oldTab);target.removeEventListener('blur',cleanup)};
    target.addEventListener('blur',cleanup,{once:true});
    setTimeout(()=>target.classList.remove('fvvt-story-focus'),1600);
  },0);
}
opener.addEventListener('click',openTrail);
})();
