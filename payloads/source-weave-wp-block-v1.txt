<script id="fv-source-weave-v1-js">
(()=>{
  if(window.__fvSourceWeave)return;
  window.__fvSourceWeave=true;

  const init=()=>{
    if(typeof HTMLDialogElement==='undefined')return;
    const article=document.querySelector('article')||document.querySelector('.wp-block-post-content')||document.querySelector('main');
    if(!article||article.querySelector('[data-fvsw-band]'))return;
    const sourceRoot=article.querySelector('[data-fv-footnotes]');
    const sourceHeading=article.querySelector('h2#sources.fv-source-heading,h2#sources,h2.fv-source-heading');
    if(!sourceRoot||!sourceHeading||!article.contains(sourceRoot)||!article.contains(sourceHeading))return;
    if(!(sourceHeading.compareDocumentPosition(sourceRoot)&Node.DOCUMENT_POSITION_FOLLOWING))return;

    const storyH2=[...article.querySelectorAll('h2')].filter(h=>h!==sourceHeading&&(h.compareDocumentPosition(sourceHeading)&Node.DOCUMENT_POSITION_FOLLOWING));
    if(storyH2.length<2||storyH2.length>8)return;
    const firstStory=storyH2[0];
    const refs=[...article.querySelectorAll('.fv-fn-ref[data-fv-fn]')].filter(ref=>!sourceRoot.contains(ref)&&(ref.compareDocumentPosition(sourceHeading)&Node.DOCUMENT_POSITION_FOLLOWING));
    if(refs.length<6||refs.length>120)return;

    const sourceItems=new Map();
    [...sourceRoot.querySelectorAll(':scope > li[id^="fv-fn-"]')].forEach(li=>{
      const match=li.id.match(/^fv-fn-(\d+)$/);
      if(match)sourceItems.set(match[1],li);
    });
    if(sourceItems.size<4||sourceItems.size>40)return;

    const sections=[];
    const openingRefs=refs.filter(ref=>ref.compareDocumentPosition(firstStory)&Node.DOCUMENT_POSITION_FOLLOWING);
    if(openingRefs.length)sections.push({key:'opening',label:'Opening',heading:null});
    storyH2.forEach((heading,index)=>sections.push({key:`section-${index+1}`,label:(heading.textContent||`Section ${index+1}`).trim(),heading}));
    if(sections.length>9)return;

    const findSection=(ref)=>{
      if(ref.compareDocumentPosition(firstStory)&Node.DOCUMENT_POSITION_FOLLOWING)return sections.find(s=>s.key==='opening')||null;
      for(let i=storyH2.length-1;i>=0;i--){
        const heading=storyH2[i];
        if(heading===ref||heading.compareDocumentPosition(ref)&Node.DOCUMENT_POSITION_FOLLOWING){
          return sections.find(s=>s.heading===heading)||null;
        }
      }
      return null;
    };

    const map=new Map();
    refs.forEach(ref=>{
      const id=String(ref.dataset.fvFn||'').trim();
      const source=sourceItems.get(id);
      const section=findSection(ref);
      if(!/^\d+$/.test(id)||!source||!section)return;
      if(!map.has(id))map.set(id,new Map());
      const bySection=map.get(id);
      if(!bySection.has(section.key))bySection.set(section.key,[]);
      bySection.get(section.key).push(ref);
    });
    const sourceIds=[...map.keys()].sort((a,b)=>Number(a)-Number(b));
    const relationCount=sourceIds.reduce((sum,id)=>sum+map.get(id).size,0);
    if(sourceIds.length<3||relationCount<5)return;

    const sourceText=(id)=>{
      const clone=sourceItems.get(id).cloneNode(true);
      clone.querySelectorAll('.fv-fn-backlinks,.fv-inline-save,.fv-evidence-trace-tools,.fvsw-band,.fvsw-dialog').forEach(el=>el.remove());
      const text=(clone.textContent||'').replace(/\s+/g,' ').trim();
      return text.length>96?`${text.slice(0,93)}…`:text;
    };
    const reused=id=>map.get(id).size>1;
    let mode='all';
    let opener=null;

    const band=document.createElement('section');
    band.className='fvsw-band';
    band.dataset.fvswBand='1';
    band.setAttribute('aria-label','Source Weave');
    const bandCopy=document.createElement('div');
    bandCopy.className='fvsw-band-copy';
    const kicker=document.createElement('p');kicker.className='fvsw-kicker';kicker.textContent='SOURCE WEAVE';
    const summary=document.createElement('p');summary.className='fvsw-summary';summary.textContent=`${sourceIds.length} cited source notes · ${sections.length} story areas · ${relationCount} source-to-story connections`;
    const boundary=document.createElement('span');boundary.className='fvsw-note';boundary.textContent='Connections reflect this page’s existing footnote links. They do not score evidence quality or confidence.';
    summary.appendChild(boundary);bandCopy.append(kicker,summary);
    const open=document.createElement('button');open.type='button';open.className='fvsw-open';open.textContent='Map the sources';
    band.append(bandCopy,open);firstStory.insertAdjacentElement('beforebegin',band);

    const dialog=document.createElement('dialog');dialog.className='fvsw-dialog';dialog.setAttribute('aria-labelledby','fvsw-title');
    const shell=document.createElement('div');shell.className='fvsw-shell';
    const head=document.createElement('header');head.className='fvsw-head';
    const headCopy=document.createElement('div');headCopy.className='fvsw-head-copy';
    const title=document.createElement('h2');title.className='fvsw-title';title.id='fvsw-title';title.textContent='Source Weave';
    const deck=document.createElement('p');deck.className='fvsw-deck';deck.textContent='See where this investigation links its published source notes across the story. A number is the count of existing footnote links in that story area—not a strength, quality or confidence score.';
    headCopy.append(title,deck);
    const close=document.createElement('button');close.type='button';close.className='fvsw-close';close.setAttribute('aria-label','Close Source Weave');close.textContent='×';
    head.append(headCopy,close);
    const controls=document.createElement('div');controls.className='fvsw-controls';
    const filters=document.createElement('div');filters.className='fvsw-filters';filters.setAttribute('aria-label','Source filters');
    const all=document.createElement('button');all.type='button';all.className='fvsw-filter';all.dataset.mode='all';all.setAttribute('aria-pressed','true');all.textContent='All cited sources';
    const repeat=document.createElement('button');repeat.type='button';repeat.className='fvsw-filter';repeat.dataset.mode='reused';repeat.setAttribute('aria-pressed','false');repeat.textContent='Reused across story areas';
    filters.append(all,repeat);
    const stats=document.createElement('div');stats.className='fvsw-stats';stats.setAttribute('aria-live','polite');
    controls.append(filters,stats);
    const body=document.createElement('div');body.className='fvsw-body';
    const legend=document.createElement('p');legend.className='fvsw-legend';legend.textContent='Select a connection to return to its first citation in that story area. Select a source label to inspect the published source note.';
    const tableWrap=document.createElement('div');tableWrap.className='fvsw-table-wrap';
    const table=document.createElement('table');table.className='fvsw-table';
    const caption=document.createElement('caption');caption.textContent='Published source notes by story area';table.appendChild(caption);
    const thead=document.createElement('thead');const trh=document.createElement('tr');
    const sourceCol=document.createElement('th');sourceCol.scope='col';sourceCol.className='fvsw-colhead fvsw-source-head';sourceCol.textContent='Published source note';trh.appendChild(sourceCol);
    sections.forEach(section=>{const th=document.createElement('th');th.scope='col';th.className='fvsw-colhead';th.textContent=section.label;trh.appendChild(th)});thead.appendChild(trh);table.appendChild(thead);
    const tbody=document.createElement('tbody');table.appendChild(tbody);tableWrap.appendChild(table);
    const mobile=document.createElement('div');mobile.className='fvsw-mobile';mobile.setAttribute('aria-label','Published source notes by story area');
    const empty=document.createElement('div');empty.className='fvsw-empty-state';empty.hidden=true;empty.textContent='No cited source notes match this view.';
    body.append(legend,tableWrap,mobile,empty);shell.append(head,controls,body);dialog.appendChild(shell);document.body.appendChild(dialog);

    const flashAndFocus=(target)=>{
      target.classList.add('fvsw-jump-target');
      const hadTab=target.hasAttribute('tabindex');const oldTab=target.getAttribute('tabindex');
      if(!target.matches('a,button,input,select,textarea,[tabindex]'))target.setAttribute('tabindex','-1');
      target.scrollIntoView({behavior:matchMedia('(prefers-reduced-motion: reduce)').matches?'auto':'smooth',block:'center'});
      target.focus({preventScroll:true});
      setTimeout(()=>{target.classList.remove('fvsw-jump-target');if(!hadTab)target.removeAttribute('tabindex');else if(oldTab!==null)target.setAttribute('tabindex',oldTab)},1600);
    };
    const closeDialog=(returnFocus=true)=>{if(dialog.open)dialog.close();if(returnFocus&&opener&&document.contains(opener))opener.focus()};
    const jumpToRef=(ref)=>{closeDialog(false);flashAndFocus(ref)};
    const jumpToSource=(id)=>{const target=sourceItems.get(id);if(!target)return;closeDialog(false);flashAndFocus(target)};

    const makeLink=(id,section)=>{
      const refsFor=map.get(id).get(section.key)||[];
      if(!refsFor.length){const dash=document.createElement('span');dash.className='fvsw-empty';dash.setAttribute('aria-hidden','true');dash.textContent='—';return dash}
      const button=document.createElement('button');button.type='button';button.className='fvsw-link';button.setAttribute('aria-label',`Source ${id} linked ${refsFor.length} time${refsFor.length===1?'':'s'} in ${section.label}; go to first citation`);
      const number=document.createElement('strong');number.textContent=String(refsFor.length);const label=document.createElement('span');label.textContent=refsFor.length===1?'citation':'citations';button.append(number,label);button.addEventListener('click',()=>jumpToRef(refsFor[0]));return button;
    };
    const makeSourceButton=(id)=>{const b=document.createElement('button');b.type='button';b.className='fvsw-source-jump';b.setAttribute('aria-label',`View published source note ${id}`);b.textContent=`Source ${id}`;const snip=document.createElement('span');snip.className='fvsw-source-snippet';snip.textContent=sourceText(id);b.appendChild(snip);b.addEventListener('click',()=>jumpToSource(id));return b};

    sourceIds.forEach(id=>{
      const row=document.createElement('tr');row.className='fvsw-row';row.dataset.sourceId=id;row.dataset.reused=reused(id)?'1':'0';
      const th=document.createElement('th');th.scope='row';th.className='fvsw-source-head';th.appendChild(makeSourceButton(id));row.appendChild(th);
      sections.forEach(section=>{const td=document.createElement('td');td.className='fvsw-cell';td.appendChild(makeLink(id,section));row.appendChild(td)});tbody.appendChild(row);
    });
    sections.forEach(section=>{
      const block=document.createElement('section');block.className='fvsw-mobile-section';block.dataset.sectionKey=section.key;
      const h=document.createElement('h3');h.className='fvsw-mobile-title';h.textContent=section.label;block.appendChild(h);
      sourceIds.forEach(id=>{
        const refsFor=map.get(id).get(section.key)||[];if(!refsFor.length)return;
        const item=document.createElement('div');item.className='fvsw-mobile-item';item.dataset.sourceId=id;item.dataset.reused=reused(id)?'1':'0';
        const source=makeSourceButton(id);source.classList.add('fvsw-mobile-source');item.append(source,makeLink(id,section));block.appendChild(item);
      });
      if(block.children.length>1)mobile.appendChild(block);
    });

    const renderMode=()=>{
      const isReused=mode==='reused';let visible=0;
      [...tbody.querySelectorAll('.fvsw-row')].forEach(row=>{const show=!isReused||row.dataset.reused==='1';row.hidden=!show;if(show)visible++});
      [...mobile.querySelectorAll('.fvsw-mobile-item')].forEach(item=>item.hidden=isReused&&item.dataset.reused!=='1');
      [...mobile.querySelectorAll('.fvsw-mobile-section')].forEach(section=>{const any=[...section.querySelectorAll('.fvsw-mobile-item')].some(item=>!item.hidden);section.hidden=!any});
      all.setAttribute('aria-pressed',isReused?'false':'true');repeat.setAttribute('aria-pressed',isReused?'true':'false');
      stats.textContent=isReused?`${visible} source${visible===1?'':'s'} appear in more than one story area`:`${visible} cited source${visible===1?'':'s'} · ${relationCount} source-to-story connections`;
      empty.hidden=visible!==0;
    };
    filters.addEventListener('click',event=>{const button=event.target.closest('.fvsw-filter');if(!button)return;mode=button.dataset.mode==='reused'?'reused':'all';renderMode()});
    open.addEventListener('click',()=>{opener=open;renderMode();dialog.showModal();close.focus()});
    close.addEventListener('click',()=>closeDialog(true));
    dialog.addEventListener('cancel',event=>{event.preventDefault();closeDialog(true)});
    dialog.addEventListener('click',event=>{if(event.target===dialog)closeDialog(true)});
    renderMode();
  };

  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',init,{once:true});
  else init();
})();
</script>
