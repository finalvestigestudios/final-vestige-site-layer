<script id="fv-case-search-v1-js">
(()=>{if(window.__fvCaseSearch)return;window.__fvCaseSearch=true;
const body=document.body;if(!body||(!body.classList.contains('single-post')&&!body.classList.contains('single')))return;
const main=document.querySelector('main')||document;const article=main.querySelector('article')||main;
const sourceRoot=article.querySelector('[data-fv-footnotes],.fv-source-list')||document.querySelector('[data-fv-footnotes],.fv-source-list');
if(!article||!sourceRoot||!('HTMLDialogElement'in window))return;
const beforeSource=node=>Boolean(node.compareDocumentPosition(sourceRoot)&Node.DOCUMENT_POSITION_FOLLOWING);
const featureUi=node=>Boolean(node.closest('.fv-case-search,.fv-passage-kit,.fv-passage-kit-action,.fv-return-trail,.fv-reading-focus,.fv-evidence-trace,.fv-source-kit,.fv-chronology-rail,.fv-evidence-map,.fv-detail-lens'));
const prose=[...article.querySelectorAll('p')].filter(p=>beforeSource(p)&&!featureUi(p)&&!p.closest('figure,figcaption,header,footer,nav,form,.sharedaddy,.sd-sharing-enabled,.wp-block-buttons')&&p.textContent.trim().length>=60);
const totalText=prose.reduce((n,p)=>n+p.textContent.trim().length,0);
const sources=[...sourceRoot.querySelectorAll('li[id^="fv-fn-"]')];
const figures=[...article.querySelectorAll('figure.wp-block-image')].filter(beforeSource);
if(prose.length<12||totalText<3000||sources.length<2)return;
const clean=s=>String(s||'').replace(/\s+/g,' ').trim();
const normalize=s=>clean(s).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLocaleLowerCase();
const headings=[...article.querySelectorAll('h2,h3')].filter(h=>beforeSource(h)&&!featureUi(h));
const sectionFor=block=>{let label='Opening';for(const h of headings){if(h===block)break;if(h.compareDocumentPosition(block)&Node.DOCUMENT_POSITION_FOLLOWING)label=clean(h.textContent)}return label};
const sourceText=li=>{const clone=li.cloneNode(true);clone.querySelectorAll('.fv-fn-backlinks,.fv-fn-back,.fv-source-kit').forEach(n=>n.remove());return clean(clone.textContent)};
let index=null;
const buildIndex=()=>{if(index)return index;const items=[];
headings.forEach((el,i)=>{const text=clean(el.textContent);if(text)items.push({scope:'story',kind:'SECTION',context:'Story section',text,target:el,order:i})});
prose.forEach((el,i)=>{const text=clean(el.textContent);if(text)items.push({scope:'story',kind:'STORY',context:sectionFor(el),text,target:el,order:100+i})});
sources.forEach((el,i)=>{const text=sourceText(el);if(text){const num=(el.id.match(/fv-fn-(\d+)/)||[])[1]||String(i+1);items.push({scope:'sources',kind:`SOURCE ${num}`,context:'Sources & Further Reading',text,target:el,order:1000+i})}});
figures.forEach((el,i)=>{const caption=clean(el.querySelector('figcaption')?.textContent);const alt=clean(el.querySelector('img')?.getAttribute('alt'));const text=clean([caption,alt].filter(Boolean).join(' — '));if(text)items.push({scope:'figures',kind:'FIGURE',context:sectionFor(el)||`Figure ${i+1}`,text,target:el,order:2000+i})});
index=items.map(item=>({...item,search:normalize(item.text)}));return index};
const launcher=document.createElement('section');launcher.className='fv-case-search';launcher.setAttribute('role','region');launcher.setAttribute('aria-label','Case Search');
launcher.innerHTML=`<div><p class="fv-case-search__kicker">CASE SEARCH</p><p class="fv-case-search__title">Find a detail without losing the thread.</p><p class="fv-case-search__note">Search this investigation’s story, ${sources.length} source ${sources.length===1?'note':'notes'}, and ${figures.length} figure ${figures.length===1?'caption':'captions'}.</p></div><button class="fv-case-search__open" type="button">Search this investigation</button>`;
const firstHeading=headings[0];if(firstHeading)firstHeading.before(launcher);else prose[Math.min(2,prose.length-1)].after(launcher);
const dialog=document.createElement('dialog');dialog.className='fv-case-search-dialog';dialog.setAttribute('aria-labelledby','fv-case-search-title');
dialog.innerHTML=`<div class="fv-case-search__panel"><div class="fv-case-search__head"><div><p class="fv-case-search__kicker">CASE SEARCH</p><h2 class="fv-case-search__dialog-title" id="fv-case-search-title">Search this investigation</h2><p class="fv-case-search__dialog-note">Find exact words in the story, source notes, and figure captions. No search query is sent anywhere.</p></div><button class="fv-case-search__close" type="button" aria-label="Close Case Search">×</button></div><div class="fv-case-search__query-wrap"><label class="fv-case-search__status" for="fv-case-search-input">Search this investigation</label><input id="fv-case-search-input" class="fv-case-search__query" type="search" autocomplete="off" spellcheck="false" placeholder="Person, place, date, detail…"></div><div class="fv-case-search__scopes" aria-label="Search scope"><button class="fv-case-search__scope" type="button" data-scope="all" aria-pressed="true">All</button><button class="fv-case-search__scope" type="button" data-scope="story" aria-pressed="false">Story</button><button class="fv-case-search__scope" type="button" data-scope="sources" aria-pressed="false">Sources</button><button class="fv-case-search__scope" type="button" data-scope="figures" aria-pressed="false">Figures</button></div><div class="fv-case-search__results" role="region" aria-label="Search results"><p class="fv-case-search__empty">Search the published page. Browser Find still works normally.</p></div><p class="fv-case-search__status" role="status" aria-live="polite"></p></div>`;
document.body.appendChild(dialog);
const open=launcher.querySelector('.fv-case-search__open');const close=dialog.querySelector('.fv-case-search__close');const input=dialog.querySelector('.fv-case-search__query');const results=dialog.querySelector('.fv-case-search__results');const status=dialog.querySelector('[role="status"]');const scopes=[...dialog.querySelectorAll('.fv-case-search__scope')];
let scope='all';let timer=0;let lastQuery='';let jumped=false;
const esc=s=>s.replace(/[.*+?^${}()|[\]\\]/g,'\\$&');
const renderSnippet=(text,tokens)=>{const frag=document.createDocumentFragment();const re=new RegExp(`(${tokens.map(esc).sort((a,b)=>b.length-a.length).join('|')})`,'ig');const first=text.search(re);const start=Math.max(0,first>70?first-70:0);const end=Math.min(text.length,start+230);const slice=(start?'…':'')+text.slice(start,end)+(end<text.length?'…':'');let pos=0;for(const m of slice.matchAll(re)){if(m.index>pos)frag.append(document.createTextNode(slice.slice(pos,m.index)));const mark=document.createElement('mark');mark.textContent=m[0];frag.append(mark);pos=m.index+m[0].length}if(pos<slice.length)frag.append(document.createTextNode(slice.slice(pos)));return frag};
const announce=text=>{status.textContent='';requestAnimationFrame(()=>status.textContent=text)};
const empty=text=>{results.replaceChildren();const p=document.createElement('p');p.className='fv-case-search__empty';p.textContent=text;results.appendChild(p)};
const search=()=>{clearTimeout(timer);const q=clean(input.value);lastQuery=q;if(q.length<2){empty(q?'Type at least two characters.':'Search the published page. Browser Find still works normally.');announce('Enter at least two characters to search.');return}const rawTokens=q.split(/\s+/).filter(Boolean);const tokens=rawTokens.map(normalize);const matches=buildIndex().filter(item=>(scope==='all'||item.scope===scope)&&tokens.every(token=>item.search.includes(token))).sort((a,b)=>a.order-b.order);results.replaceChildren();if(!matches.length){empty(`No ${scope==='all'?'page':scope} matches for “${q}”.`);announce(`No results for ${q}.`);return}matches.slice(0,60).forEach(item=>{const button=document.createElement('button');button.type='button';button.className='fv-case-search__result';const kind=document.createElement('span');kind.className='fv-case-search__kind';kind.textContent=item.kind;const body=document.createElement('span');const context=document.createElement('span');context.className='fv-case-search__context';context.textContent=item.context;const snippet=document.createElement('span');snippet.className='fv-case-search__snippet';snippet.appendChild(renderSnippet(item.text,rawTokens));body.append(context,snippet);button.append(kind,body);button.addEventListener('click',()=>jump(item));results.appendChild(button)});const shown=Math.min(matches.length,60);announce(`${matches.length} ${matches.length===1?'result':'results'} for ${q}.${matches.length>shown?` Showing first ${shown}.`:''}`)};
const jump=item=>{jumped=true;dialog.close();requestAnimationFrame(()=>{const target=item.target;const reduce=matchMedia('(prefers-reduced-motion: reduce)').matches;target.scrollIntoView({behavior:reduce?'auto':'smooth',block:'center'});const hadTab=target.hasAttribute('tabindex');const oldTab=target.getAttribute('tabindex');target.setAttribute('tabindex','-1');target.classList.add('fv-case-search-target');target.focus({preventScroll:true});const cleanup=()=>{target.classList.remove('fv-case-search-target');if(!hadTab)target.removeAttribute('tabindex');else target.setAttribute('tabindex',oldTab);target.removeEventListener('blur',cleanup)};target.addEventListener('blur',cleanup);setTimeout(()=>target.classList.remove('fv-case-search-target'),1800)})};
open.addEventListener('click',()=>{jumped=false;buildIndex();dialog.showModal();requestAnimationFrame(()=>{input.focus();input.select()})});
close.addEventListener('click',()=>dialog.close());
scopes.forEach(button=>button.addEventListener('click',()=>{scope=button.dataset.scope;scopes.forEach(b=>b.setAttribute('aria-pressed',b===button?'true':'false'));search();input.focus()}));
input.addEventListener('input',()=>{clearTimeout(timer);timer=setTimeout(search,80)});
input.addEventListener('keydown',e=>{if(e.key==='ArrowDown'){const first=results.querySelector('.fv-case-search__result');if(first){e.preventDefault();first.focus()}}});
results.addEventListener('keydown',e=>{if(!['ArrowDown','ArrowUp','Home','End'].includes(e.key))return;const buttons=[...results.querySelectorAll('.fv-case-search__result')];if(!buttons.length)return;const current=buttons.indexOf(document.activeElement);let next=current;if(e.key==='ArrowDown')next=Math.min(buttons.length-1,current+1);if(e.key==='ArrowUp')next=Math.max(0,current-1);if(e.key==='Home')next=0;if(e.key==='End')next=buttons.length-1;if(next>=0){e.preventDefault();buttons[next].focus()}});
dialog.addEventListener('close',()=>{clearTimeout(timer);if(!jumped)open.focus();jumped=false});
dialog.addEventListener('click',e=>{const r=dialog.getBoundingClientRect();if(e.target===dialog&&(e.clientX<r.left||e.clientX>r.right||e.clientY<r.top||e.clientY>r.bottom))dialog.close()});
})();
</script>
