<script id="fv-evidence-trace-v1-js">
(()=>{
  if(window.__fvEvidenceTrace)return;
  window.__fvEvidenceTrace=true;

  const init=()=>{
    const article=document.querySelector('article')||document.querySelector('.wp-block-post-content')||document.querySelector('main');
    if(!article||article.querySelector('.fv-evidence-trace-bar'))return;
    const sourceRoot=article.querySelector('[data-fv-footnotes]');
    if(!sourceRoot)return;
    const sourceHeading=sourceRoot.previousElementSibling&&sourceRoot.previousElementSibling.matches('h2,h3')?sourceRoot.previousElementSibling:article.querySelector('#sources,.fv-source-heading');
    const firstSection=[...article.querySelectorAll('h2')].find(h=>h!==sourceHeading&&h.compareDocumentPosition(sourceRoot)&Node.DOCUMENT_POSITION_FOLLOWING);
    if(!firstSection)return;

    const rawRefs=[...article.querySelectorAll('.fv-fn-ref[data-fv-fn]')].filter(ref=>!sourceRoot.contains(ref)&&ref.compareDocumentPosition(sourceRoot)&Node.DOCUMENT_POSITION_FOLLOWING);
    const passageMap=new Map();
    rawRefs.forEach(ref=>{
      const passage=ref.closest('p,blockquote,li');
      if(!passage||sourceRoot.contains(passage))return;
      const note=String(ref.dataset.fvFn||'').trim();
      if(!/^\d+$/.test(note))return;
      if(!passageMap.has(passage))passageMap.set(passage,new Set());
      passageMap.get(passage).add(note);
    });
    const passages=[...passageMap.entries()].filter(([,notes])=>notes.size>0);
    const noteIds=new Set(passages.flatMap(([,notes])=>[...notes]));
    const validNotes=[...noteIds].filter(note=>sourceRoot.querySelector(`#fv-fn-${note}`));
    if(passages.length<4||validNotes.length<3)return;

    const bar=document.createElement('section');
    bar.className='fv-evidence-trace-bar';
    bar.setAttribute('aria-label','Evidence trace');
    const copy=document.createElement('div');
    copy.className='fv-evidence-trace-copy';
    const kicker=document.createElement('p');
    kicker.className='fv-evidence-trace-kicker';
    kicker.textContent='EVIDENCE TRACE';
    const summary=document.createElement('p');
    summary.className='fv-evidence-trace-summary';
    summary.textContent=`${passages.length} linked passage${passages.length===1?'':'s'} · ${validNotes.length} source note${validNotes.length===1?'':'s'}`;
    const note=document.createElement('span');
    note.className='fv-evidence-trace-note';
    note.textContent='Shows where this page links to source notes. It does not score uncited text.';
    summary.appendChild(note);
    copy.append(kicker,summary);
    const toggle=document.createElement('button');
    toggle.type='button';
    toggle.className='fv-evidence-trace-toggle';
    toggle.setAttribute('aria-pressed','false');
    toggle.textContent='Show linked evidence';
    bar.append(copy,toggle);
    firstSection.insertAdjacentElement('beforebegin',bar);

    const makeCard=(noteId)=>{
      const source=sourceRoot.querySelector(`#fv-fn-${noteId}`);
      if(!source)return null;
      const clone=source.cloneNode(true);
      clone.removeAttribute('id');
      clone.querySelectorAll('.fv-fn-backlinks,.fv-inline-save,.fv-evidence-trace-tools').forEach(el=>el.remove());
      clone.querySelectorAll('[id]').forEach(el=>el.removeAttribute('id'));
      const card=document.createElement('div');
      card.className='fv-evidence-trace-card';
      const label=document.createElement('span');
      label.className='fv-evidence-trace-card-label';
      label.textContent=`Source note ${noteId}`;
      const body=document.createElement('div');
      while(clone.firstChild)body.appendChild(clone.firstChild);
      card.append(label,body);
      return card;
    };

    passages.forEach(([passage,notes],index)=>{
      passage.classList.add('fv-evidence-trace-passage');
      const tool=document.createElement('div');
      tool.className='fv-evidence-trace-tools';
      const button=document.createElement('button');
      button.type='button';
      button.className='fv-evidence-trace-chip';
      button.id=`fv-evidence-trace-trigger-${index+1}`;
      const ids=[...notes].filter(id=>validNotes.includes(id)).sort((a,b)=>Number(a)-Number(b));
      if(!ids.length)return;
      button.textContent=ids.length===1?`Source ${ids[0]}`:`Sources ${ids.join(', ')}`;
      const panel=document.createElement('div');
      panel.className='fv-evidence-trace-panel';
      panel.id=`fv-evidence-trace-panel-${index+1}`;
      panel.hidden=true;
      panel.setAttribute('role','region');
      panel.setAttribute('aria-labelledby',button.id);
      button.setAttribute('aria-controls',panel.id);
      button.setAttribute('aria-expanded','false');
      ids.forEach(id=>{const card=makeCard(id);if(card)panel.appendChild(card)});
      if(!panel.children.length){
        const empty=document.createElement('div');
        empty.className='fv-evidence-trace-empty';
        empty.textContent='Source note unavailable on this page.';
        panel.appendChild(empty);
      }
      button.addEventListener('click',()=>{
        const open=button.getAttribute('aria-expanded')==='true';
        button.setAttribute('aria-expanded',open?'false':'true');
        panel.hidden=open;
      });
      tool.append(button,panel);
      passage.insertAdjacentElement('afterend',tool);
    });

    const setActive=(active)=>{
      article.classList.toggle('fv-evidence-trace-active',active);
      toggle.setAttribute('aria-pressed',active?'true':'false');
      toggle.textContent=active?'Hide linked evidence':'Show linked evidence';
      if(!active){
        article.querySelectorAll('.fv-evidence-trace-chip[aria-expanded="true"]').forEach(button=>{
          button.setAttribute('aria-expanded','false');
          const panel=document.getElementById(button.getAttribute('aria-controls'));
          if(panel)panel.hidden=true;
        });
      }
    };
    toggle.addEventListener('click',()=>setActive(toggle.getAttribute('aria-pressed')!=='true'));
  };

  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',init,{once:true});
  else init();
})();
</script>
