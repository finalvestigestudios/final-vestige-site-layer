<script id="fv-passage-kit-v1-js">
(()=>{if(window.__fvPassageKit)return;window.__fvPassageKit=true;
const body=document.body;
if(!body||( !body.classList.contains('single-post') && !body.classList.contains('single') ))return;
const main=document.querySelector('main')||document;
const article=main.querySelector('article')||main;
const sourceRoot=document.querySelector('[data-fv-footnotes],.fv-source-list');
if(!sourceRoot||!article)return;
const beforeSource=node=>Boolean(node.compareDocumentPosition(sourceRoot)&Node.DOCUMENT_POSITION_FOLLOWING);
const excluded=p=>Boolean(p.closest('[data-fv-footnotes],.fv-source-list,figure,figcaption,header,footer,nav,form,.sharedaddy,.sd-sharing-enabled,.wp-block-buttons,.fv-fn-popover,.fv-passage-kit,.fv-passage-kit-action'));
const prose=[...article.querySelectorAll('p')].filter(p=>beforeSource(p)&&!excluded(p)&&p.textContent.trim().length>=80);
const totalText=prose.reduce((n,p)=>n+p.textContent.trim().length,0);
if(prose.length<12||totalText<3000)return;
const sourceItems=[...sourceRoot.querySelectorAll('li[id^="fv-fn-"]')];
if(sourceItems.length<2)return;
const clean=s=>String(s||'').replace(/\s+/g,' ').trim();
const articleTitle=clean(article.querySelector('h1')?.textContent||document.querySelector('h1')?.textContent||document.title);
const pageUrl=location.origin+location.pathname;
const headings=[...article.querySelectorAll('h2,h3')].filter(h=>beforeSource(h));
const sectionFor=block=>{let label='';for(const h of headings){if(h.compareDocumentPosition(block)&Node.DOCUMENT_POSITION_FOLLOWING)label=clean(h.textContent)}return label};
const sourceFor=num=>{const li=document.getElementById('fv-fn-'+num);if(!li||!sourceRoot.contains(li))return null;const clone=li.cloneNode(true);clone.querySelectorAll('.fv-fn-backlinks,.fv-fn-back').forEach(n=>n.remove());const external=[...clone.querySelectorAll('a[href]')].find(a=>!a.getAttribute('href').startsWith('#'));const url=external?.href||'';if(external)external.remove();const text=clean(clone.textContent).replace(/\s+[.,;:!?]$/,'').trim();if(!text)return null;return {num:String(num),text,url}};
const refsFor=block=>[...new Set([...block.querySelectorAll('.fv-fn-ref[data-fv-fn]')].map(a=>a.dataset.fvFn).filter(Boolean))].map(sourceFor).filter(Boolean);
const quoteFor=block=>{const clone=block.cloneNode(true);clone.querySelectorAll('.fv-fn-mark,.fv-passage-kit-action').forEach(n=>n.remove());return clean(clone.textContent)};
const format=(block,sources)=>{const quote=quoteFor(block);const section=sectionFor(block);const lines=[`“${quote}”`,'',articleTitle];if(section)lines.push(`Section: ${section}`);lines.push(pageUrl);if(sources.length){lines.push('','Linked source '+(sources.length===1?'note:':'notes:'));sources.forEach(s=>{lines.push(`[${s.num}] ${s.text}`);if(s.url)lines.push(s.url)})}return lines.join('\n')};
const bar=document.createElement('section');
bar.className='fv-passage-kit';
bar.setAttribute('role','region');
bar.setAttribute('aria-label','Passage Kit');
bar.innerHTML='<div><p class="fv-passage-kit__kicker">PASSAGE KIT</p><p class="fv-passage-kit__title">Carry a published passage with its linked source notes.</p><p class="fv-passage-kit__note">Copies published wording and linked source notes. It does not rate the evidence.</p></div><button class="fv-passage-kit__toggle" type="button" aria-pressed="false">Show copy controls</button><p class="fv-passage-kit__status" role="status" aria-live="polite"></p>';
const firstHeading=headings[0];
if(firstHeading)firstHeading.before(bar);else prose[Math.min(2,prose.length-1)].after(bar);
const toggle=bar.querySelector('.fv-passage-kit__toggle');
const status=bar.querySelector('.fv-passage-kit__status');
let built=false;
const announce=text=>{status.textContent='';requestAnimationFrame(()=>{status.textContent=text})};
const copyText=async text=>{try{if(!navigator.clipboard?.writeText)throw new Error('clipboard unavailable');await navigator.clipboard.writeText(text);return true}catch(e){window.prompt('Copy this passage and its source notes:',text);return false}};
const build=()=>{if(built)return;built=true;prose.forEach((block,index)=>{const sources=refsFor(block);const action=document.createElement('div');action.className='fv-passage-kit-action';const button=document.createElement('button');button.type='button';button.dataset.fvPassage=String(index+1);button.setAttribute('aria-label',sources.length?`Copy passage ${index+1} with ${sources.length} linked source ${sources.length===1?'note':'notes'}`:`Copy passage ${index+1}`);button.innerHTML=sources.length?`Copy passage + sources <span>${sources.length}</span>`:'Copy passage';button.addEventListener('click',async()=>{const ok=await copyText(format(block,sources));announce(ok?(sources.length?`Passage copied with ${sources.length} linked source ${sources.length===1?'note':'notes'}.`:'Passage copied.'):'Copy text is ready in the fallback dialog.')});action.appendChild(button);block.after(action)})};
const setActive=active=>{if(active)build();article.classList.toggle('fv-passage-kit-on',active);toggle.setAttribute('aria-pressed',active?'true':'false');toggle.textContent=active?'Hide copy controls':'Show copy controls';announce(active?'Passage copy controls shown.':'Passage copy controls hidden.')};
toggle.addEventListener('click',()=>setActive(toggle.getAttribute('aria-pressed')!=='true'));
document.addEventListener('keydown',e=>{if(e.key==='Escape'&&toggle.getAttribute('aria-pressed')==='true'){setActive(false);toggle.focus()}});
})();
</script>
