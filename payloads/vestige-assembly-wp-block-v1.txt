<script id="fv-vestige-assembly-v1-js">
(()=>{
if(window.__fvVestigeAssembly)return;
window.__fvVestigeAssembly=true;
try{
// FV_VESTIGE_ASSEMBLY_GATE_START
const fvVaShouldRun=(ctx)=>{
  if(!ctx)return false;
  if(ctx.reducedMotion||ctx.forcedColors)return false;
  if(ctx.sessionSeen)return false;
  const onHome=ctx.isHomeClass||ctx.pathname==='/';
  return !!onHome;
};
const fvVaTimeline=()=>[
  {stage:'run',delay:0},
  {stage:'connect',delay:4400},
  {stage:'assemble',delay:5600},
  {stage:'fade',delay:7600},
  {stage:'remove',delay:8600}
];
// FV_VESTIGE_ASSEMBLY_GATE_END

if(typeof document==='undefined'||!document.body){window.__fvVestigeAssembly=false;return}

const SEEN_KEY='fvVestigeAssemblySeenV1';
let sessionSeen=false;
try{sessionSeen=sessionStorage.getItem(SEEN_KEY)==='1'}catch(_){sessionSeen=true}

let reducedMotion=false;
try{reducedMotion=window.matchMedia('(prefers-reduced-motion: reduce)').matches}catch(_){reducedMotion=false}

let forcedColors=false;
try{forcedColors=window.matchMedia('(forced-colors: active)').matches}catch(_){forcedColors=false}

const ctx={
  reducedMotion:reducedMotion,
  forcedColors:forcedColors,
  sessionSeen:sessionSeen,
  isHomeClass:document.body.classList.contains('home'),
  pathname:location.pathname
};

if(!fvVaShouldRun(ctx))return;

try{sessionStorage.setItem(SEEN_KEY,'1')}catch(_){/* private mode: acceptable, may replay once more */}

const FRAGMENTS=[
  {type:'document',tag:'Document',line:'Buried in a footnote.'},
  {type:'files',tag:'Record set',line:'Split across six files.'},
  {type:'map',tag:'Map',line:'Marked on a map nobody thought to compare.'},
  {type:'photograph',tag:'Photograph',line:'Left in a photograph taken eighty years ago.'}
];

const root=document.createElement('div');
root.id='fv-vestige-assembly';
root.setAttribute('aria-hidden','true');
root.setAttribute('role','presentation');
if('inert' in root)root.inert=true;

const noise=document.createElement('div');
noise.className='fv-va-noise';
root.appendChild(noise);

const stage=document.createElement('div');
stage.className='fv-va-stage';
FRAGMENTS.forEach((f,i)=>{
  const card=document.createElement('div');
  card.className='fv-va-frag';
  card.setAttribute('data-fv-va-i',String(i));
  const art=document.createElement('div');
  art.className='fv-va-art';
  art.setAttribute('data-fv-va-type',f.type);
  const tag=document.createElement('div');
  tag.className='fv-va-tag';
  tag.textContent=f.tag;
  const line=document.createElement('div');
  line.className='fv-va-line';
  line.textContent=f.line;
  card.append(art,tag,line);
  stage.appendChild(card);
});
root.appendChild(stage);

const rule=document.createElement('div');
rule.className='fv-va-rule';
root.appendChild(rule);

const mark=document.createElement('div');
mark.className='fv-va-mark';
mark.append(document.createElement('span'),document.createElement('span'),document.createElement('span'));
root.appendChild(mark);

document.body.appendChild(root);

const timers=[];
let done=false;

const clearTimers=()=>{
  timers.forEach((t)=>clearTimeout(t));
  timers.length=0;
};

const finish=()=>{
  if(done)return;
  done=true;
  clearTimers();
  window.removeEventListener('scroll',dismiss);
  window.removeEventListener('keydown',dismiss);
  window.removeEventListener('pointerdown',dismiss);
  if(root.parentNode)root.parentNode.removeChild(root);
};

const dismiss=()=>{
  if(done)return;
  clearTimers();
  root.setAttribute('data-fv-va-fast','1');
  root.setAttribute('data-fv-va-fade','1');
  timers.push(setTimeout(finish,320));
};

window.addEventListener('scroll',dismiss,{passive:true});
window.addEventListener('keydown',dismiss);
window.addEventListener('pointerdown',dismiss,{passive:true});

fvVaTimeline().forEach((step)=>{
  if(step.stage==='run'){
    timers.push(setTimeout(()=>requestAnimationFrame(()=>root.setAttribute('data-fv-va-run','1')),step.delay));
  }else if(step.stage==='connect'){
    timers.push(setTimeout(()=>root.setAttribute('data-fv-va-connect','1'),step.delay));
  }else if(step.stage==='assemble'){
    timers.push(setTimeout(()=>root.setAttribute('data-fv-va-assemble','1'),step.delay));
  }else if(step.stage==='fade'){
    timers.push(setTimeout(()=>root.setAttribute('data-fv-va-fade','1'),step.delay));
  }else if(step.stage==='remove'){
    timers.push(setTimeout(finish,step.delay));
  }
});

timers.push(setTimeout(finish,12000));

}catch(_){
  window.__fvVestigeAssembly=false;
  const stray=document.getElementById('fv-vestige-assembly');
  if(stray&&stray.parentNode)stray.parentNode.removeChild(stray);
}
})();
</script>
