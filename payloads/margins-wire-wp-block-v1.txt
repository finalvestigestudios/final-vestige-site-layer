<script id="fv-margins-wire-v1-js">
(()=>{
if(window.__fvMarginsWire)return;
window.__fvMarginsWire=true;
try{
// FV_MARGINS_WIRE_GATE_START
const fvMwShouldRun=(items)=>Array.isArray(items)&&items.length>1;
// FV_MARGINS_WIRE_GATE_END

const root=document.querySelector('.fv-margin-strip');
const inner=root&&root.querySelector('.fv-margin-strip__inner');
const win=root&&root.querySelector('.fv-margin-strip__window');
const items=win?Array.from(win.querySelectorAll('.fv-margin-strip__item')):[];

if(!fvMwShouldRun(items)){window.__fvMarginsWire=false;return}

let reducedMotion=false;
try{reducedMotion=window.matchMedia('(prefers-reduced-motion: reduce)').matches}catch(_){reducedMotion=false}

root.classList.add('fv-margin-strip--wired');

const controls=document.createElement('div');
controls.className='fv-margin-strip__controls';
const prevBtn=document.createElement('button');
prevBtn.type='button';prevBtn.className='fv-margin-strip__ctrl-btn';prevBtn.setAttribute('aria-label','Previous fragment');prevBtn.textContent='‹';
const playBtn=document.createElement('button');
playBtn.type='button';playBtn.className='fv-margin-strip__ctrl-btn';playBtn.setAttribute('aria-label','Pause rotation');playBtn.textContent='⏸';
const nextBtn=document.createElement('button');
nextBtn.type='button';nextBtn.className='fv-margin-strip__ctrl-btn';nextBtn.setAttribute('aria-label','Next fragment');nextBtn.textContent='›';
const count=document.createElement('span');
count.className='fv-margin-strip__ctrl-count';
controls.append(prevBtn,playBtn,nextBtn,count);
inner.appendChild(controls);

let index=0;
let playing=!reducedMotion;
let timer=null;

const paint=()=>{
  items.forEach((el,i)=>{
    el.style.setProperty('animation','none','important');
    el.style.setProperty('filter','none','important');
    el.style.setProperty('display','block','important');
    if(i===index){
      el.style.setProperty('position','relative','important');
      el.style.setProperty('opacity','1','important');
      el.style.setProperty('transform','none','important');
      el.setAttribute('aria-hidden','false');
    }else{
      el.style.setProperty('position','absolute','important');
      el.style.setProperty('opacity','0','important');
      el.setAttribute('aria-hidden','true');
    }
  });
  count.textContent=String(index+1).padStart(2,'0')+' / '+items.length;
};

const show=(i)=>{index=(i+items.length)%items.length;paint()};

const stop=()=>{if(timer){clearInterval(timer);timer=null}};
const start=()=>{stop();if(!playing)return;timer=setInterval(()=>show(index+1),5200)};

paint();
start();

const pauseForInteraction=()=>{
  if(!playing)return;
  playing=false;stop();
  playBtn.textContent='▶';
  playBtn.setAttribute('aria-label','Resume rotation');
};
win.addEventListener('mouseenter',pauseForInteraction);
win.addEventListener('focusin',pauseForInteraction);
win.setAttribute('tabindex','-1');

win.addEventListener('keydown',(e)=>{
  if(e.key==='ArrowRight'){pauseForInteraction();show(index+1)}
  else if(e.key==='ArrowLeft'){pauseForInteraction();show(index-1)}
});
prevBtn.addEventListener('click',()=>{pauseForInteraction();show(index-1)});
nextBtn.addEventListener('click',()=>{pauseForInteraction();show(index+1)});
playBtn.addEventListener('click',()=>{
  playing=!playing;
  playBtn.textContent=playing?'⏸':'▶';
  playBtn.setAttribute('aria-label',playing?'Pause rotation':'Resume rotation');
  if(playing)start();else stop();
});

}catch(_){
  window.__fvMarginsWire=false;
}
})();
</script>
