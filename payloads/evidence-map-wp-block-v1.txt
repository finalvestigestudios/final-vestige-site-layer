<script id="fv-evidence-map-v1-js">
(()=>{
  if(window.__fvEvidenceMap)return;
  window.__fvEvidenceMap=true;
  const ready=fn=>document.readyState==='loading'?document.addEventListener('DOMContentLoaded',fn,{once:true}):fn();
  ready(()=>{
    try{
      const root=document.querySelector('main');
      const launch=document.querySelector('[data-fv-em-launch]');
      const dialog=document.querySelector('[data-fv-em-dialog]');
      if(!root||!launch||!dialog||typeof dialog.showModal!=='function')return;

      const headings=[...root.querySelectorAll('h2')];
      const sourceHeading=headings.find(h=>h.id==='sources'||/sources\s*(?:&|and)\s*(?:further\s*)?reading|sources$/i.test(h.textContent.trim()));
      if(!sourceHeading)return;

      const FOLLOWING=Node.DOCUMENT_POSITION_FOLLOWING;
      const before=(a,b)=>!!(a.compareDocumentPosition(b)&FOLLOWING);
      const findSourceList=()=>{
        const direct=root.querySelector('.fv-source-list[data-fv-footnotes]');
        if(direct&&before(sourceHeading,direct))return direct;
        let n=sourceHeading.nextElementSibling,steps=0;
        while(n&&steps++<8){
          if(n.matches('ul,ol'))return n;
          const inner=n.querySelector&&n.querySelector('ul,ol');
          if(inner)return inner;
          n=n.nextElementSibling;
        }
        return null;
      };
      const sourceList=findSourceList();
      const sourceItems=sourceList?[...sourceList.querySelectorAll(':scope > li')]:[];
      if(sourceItems.length<2)return;

      const sections=headings.filter(h=>h!==sourceHeading&&before(h,sourceHeading)&&!/^(Evidence Notes|Sources(?:\s*&|\s+and)?\s*(?:Further\s*)?Reading)$/i.test(h.textContent.trim()));
      if(sections.length<2)return;

      const allRefs=[...root.querySelectorAll('a[href^="#"]')].filter(a=>before(a,sourceHeading)&&(/fv-fn-|footnote|fnref|fn-/i.test(a.getAttribute('href')||'')||a.closest('sup.fn,.fv-fn-mark')));
      const figures=[...root.querySelectorAll('figure')].filter(f=>before(f,sourceHeading));
      const statBox=dialog.querySelector('[data-fv-em-stats]');
      const storyList=dialog.querySelector('[data-fv-em-story]');
      const sourceIndex=dialog.querySelector('[data-fv-em-sources]');
      const state=launch.querySelector('[data-fv-em-state]');
      const closeBtn=dialog.querySelector('[data-fv-em-close]');
      const tabs=[...dialog.querySelectorAll('[data-fv-em-tab]')];
      let opener=null,navigating=false,currentSection=0;

      const ensureId=(el,prefix,i)=>{if(!el.id)el.id=`${prefix}-${i+1}`;return el.id};
      sections.forEach((h,i)=>ensureId(h,'fv-evidence-section',i));
      sourceItems.forEach((li,i)=>ensureId(li,'fv-evidence-source',i));

      const sectionBounds=sections.map((h,i)=>({h,next:sections[i+1]||sourceHeading}));
      const inRange=(node,start,end)=>before(start,node)&&before(node,end);
      const sectionData=sectionBounds.map(({h,next},i)=>({
        el:h,
        title:h.textContent.trim(),
        citations:allRefs.filter(n=>inRange(n,h,next)).length,
        figures:figures.filter(n=>inRange(n,h,next)).length,
        i
      }));

      const sourceRefCount=li=>{
        if(!li.id)return 0;
        const escaped=CSS.escape(li.id);
        return root.querySelectorAll(`a[href="#${escaped}"]`).length;
      };
      const domainFor=li=>{
        const a=[...li.querySelectorAll('a[href]')].find(x=>{try{return new URL(x.href,location.href).origin!==location.origin}catch(_){return false}});
        if(!a)return '';
        try{return new URL(a.href,location.href).hostname.replace(/^www\./,'')}catch(_){return ''}
      };
      const cleanSourceTitle=(li,i)=>{
        const clone=li.cloneNode(true);
        clone.querySelectorAll('.fv-fn-backlinks,.fv-fn-back').forEach(n=>n.remove());
        const t=clone.textContent.replace(/\s+/g,' ').trim();
        return t.length>118?t.slice(0,115).trim()+'…':(t||`Source ${i+1}`);
      };
      const sourceData=sourceItems.map((li,i)=>({el:li,title:cleanSourceTitle(li,i),domain:domainFor(li),refs:sourceRefCount(li),i}));

      statBox.innerHTML=[['Sections',sections.length],['Citations',allRefs.length],['Figures',figures.length],['Sources',sourceItems.length]].map(([label,value])=>`<div class="fv-em-stat"><b>${value}</b><span>${label}</span></div>`).join('');

      const escapeHtml=s=>String(s).replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
      const drawStory=()=>{
        storyList.innerHTML=sectionData.map(d=>`<li><button class="fv-em-row" type="button" data-fv-em-jump="section" data-index="${d.i}" aria-current="${d.i===currentSection?'true':'false'}"><span><span class="fv-em-row__title">${escapeHtml(d.title)}</span><span class="fv-em-row__meta">${d.citations} citation${d.citations===1?'':'s'} · ${d.figures} figure${d.figures===1?'':'s'}</span></span><span class="fv-em-row__count">${d.i+1}/${sections.length}</span></button></li>`).join('');
      };
      const drawSources=()=>{
        sourceIndex.innerHTML=sourceData.map(d=>`<li><button class="fv-em-row" type="button" data-fv-em-jump="source" data-index="${d.i}"><span><span class="fv-em-row__title">${escapeHtml(d.title)}</span><span class="fv-em-row__meta">${escapeHtml(d.domain||'Source record')}</span></span><span class="fv-em-row__count">${d.refs?`${d.refs} cite${d.refs===1?'':'s'}`:`Source ${d.i+1}`}</span></button></li>`).join('');
      };
      const updateLaunch=()=>{state.textContent=`${currentSection+1}/${sections.length} · ${sourceItems.length} sources`};
      drawStory();drawSources();updateLaunch();launch.dataset.ready='true';

      const setTab=(name,focus=false)=>{
        tabs.forEach(t=>{const on=t.dataset.fvEmTab===name;t.setAttribute('aria-selected',String(on));t.tabIndex=on?0:-1;if(on&&focus)t.focus()});
        dialog.querySelectorAll('[data-fv-em-panel]').forEach(p=>p.hidden=p.dataset.fvEmPanel!==name);
      };
      tabs.forEach((tab,i)=>{
        tab.addEventListener('click',()=>setTab(tab.dataset.fvEmTab));
        tab.addEventListener('keydown',e=>{
          if(!['ArrowLeft','ArrowRight','Home','End'].includes(e.key))return;
          e.preventDefault();
          let next=i;
          if(e.key==='ArrowLeft')next=(i-1+tabs.length)%tabs.length;
          if(e.key==='ArrowRight')next=(i+1)%tabs.length;
          if(e.key==='Home')next=0;
          if(e.key==='End')next=tabs.length-1;
          setTab(tabs[next].dataset.fvEmTab,true);
        });
      });

      const open=from=>{opener=from||document.activeElement;drawStory();updateLaunch();dialog.showModal();closeBtn.focus()};
      const close=()=>{if(dialog.open)dialog.close()};
      launch.addEventListener('click',()=>open(launch));
      closeBtn.addEventListener('click',close);
      dialog.addEventListener('click',e=>{if(e.target!==dialog)return;const r=dialog.getBoundingClientRect();if(e.clientX<r.left||e.clientX>r.right||e.clientY<r.top||e.clientY>r.bottom)close()});
      dialog.addEventListener('close',()=>{if(!navigating&&opener&&document.contains(opener))opener.focus();opener=null});

      const jumpTo=el=>{
        if(!el)return;
        navigating=true;
        dialog.close();
        const reduce=matchMedia('(prefers-reduced-motion: reduce)').matches;
        const hadTab=el.hasAttribute('tabindex');
        if(!hadTab)el.setAttribute('tabindex','-1');
        setTimeout(()=>{
          el.scrollIntoView({behavior:reduce?'auto':'smooth',block:'center'});
          el.classList.add('fv-em-target-flash');
          el.focus({preventScroll:true});
          setTimeout(()=>el.classList.remove('fv-em-target-flash'),1200);
          const cleanup=()=>{if(!hadTab)el.removeAttribute('tabindex')};
          el.addEventListener('blur',cleanup,{once:true});
          navigating=false;opener=null;
        },reduce?0:80);
      };
      dialog.addEventListener('click',e=>{
        const btn=e.target.closest('[data-fv-em-jump]');if(!btn)return;
        const i=Number(btn.dataset.index);
        jumpTo(btn.dataset.fvEmJump==='section'?sectionData[i]?.el:sourceData[i]?.el);
      });

      if('IntersectionObserver'in window){
        const io=new IntersectionObserver(entries=>{
          const visible=entries.filter(x=>x.isIntersecting).sort((a,b)=>Math.abs(a.boundingClientRect.top)-Math.abs(b.boundingClientRect.top))[0];
          if(!visible)return;
          const i=sections.indexOf(visible.target);
          if(i>=0&&i!==currentSection){currentSection=i;drawStory();updateLaunch()}
        },{rootMargin:'-18% 0px -62% 0px',threshold:0});
        sections.forEach(h=>io.observe(h));
      }
    }catch(e){console.warn('Final Vestige Evidence Map unavailable',e)}
  });
})();
</script>
