<script id="fv-return-trail-v1-js">
(()=>{if(window.__fvReturnTrail)return;window.__fvReturnTrail=true;

if(isSingle&&root){

  const excluded='[data-fv-footnotes],.fv-source-list,figcaption,nav,footer,.wp-block-navigation,.sharedaddy,.sd-sharing-enabled,.fv-return-trail-card';
  const prose=[...root.querySelectorAll('p')].filter(p=>beforeSource(p)&&!p.closest(excluded)&&p.textContent.trim().length>=55);
  const totalText=prose.reduce((n,p)=>n+p.textContent.trim().length,0);
  if(prose.length<12||totalText<3000)return;
  const markers=[...root.querySelectorAll('h2,h3,p')].filter(el=>beforeSource(el)&&!el.closest(excluded)&&el.textContent.trim().length>=20);
  if(markers.length<10)return;
  markers.forEach((el,i)=>{el.dataset.fvReturnTrailIndex=String(i);el.classList.add('fv-return-trail-marker')});
  const title=cleanTitle(document.querySelector('h1')?.textContent||document.title.replace(/\s*[|–—-].*$/,''));
  const current=read().find(x=>x.path===path);
  if(current&&current.index>=0&&current.index<markers.length&&current.percent>=5&&current.percent<95){
    const slot=prose[Math.min(1,prose.length-1)];const ui=createCard(current,'same');slot.insertAdjacentElement('afterend',ui.card);const resume=()=>{const target=markers[current.index]||markers[0];target.scrollIntoView({block:'center',behavior:reduced()?'auto':'smooth'});target.classList.add('fv-return-trail-flash');setTimeout(()=>target.classList.remove('fv-return-trail-flash'),1400);target.setAttribute('tabindex','-1');target.focus({preventScroll:true});announce(`Returned to ${current.heading||'your last place'}, ${pct(current.percent)} through this page.`)};ui.go.addEventListener('click',resume);if(location.hash==='#fv-return-trail'){history.replaceState(null,'',path);requestAnimationFrame(()=>resume())}
  }

  const queue=()=>{if(timer)clearTimeout(timer);timer=setTimeout(snapshot,850)};

  return;
}

const ui=createCard(recent,'other');main.insertAdjacentElement('afterbegin',ui.card);
})();
